#- Parameter Definitions
(define LCon 100); Device Length(100um)
(define LD @LD@); Device Length(@LD@um)
(define WB @WB@); Device Width(600um)
(define GW @GW@) ; GaN Width (@GW@)

#- Material Definitions
#-Ga2O3
(sdegeo:create-rectangle (position 0 0 0) (position LD (+ WB 7) 0) "Ga2O3" "Ga2O3_base"); Ga2O3_base
(sdegeo:create-rectangle (position 0 0 0) (position LD 7 0) "Ga2O3" "Ga2O3_epi"); Ga2O3_epi
(sdegeo:create-rectangle (position 0 0 0) (position LCon GW 0) "GaN" "GaN_con"); GaN_con

#- Constant Doping Profile in Ga2O3 Substrate
(sdepe:doping-constant-placement "BaseDoping" "TinActiveConcentration" 2.5e18 "Ga2O3_base")
(sdepe:doping-constant-placement "EpiDoping" "SiliconActiveConcentration" @Nd@ "Ga2O3_epi")
(sdepe:doping-constant-placement "ConDoping" "MagnesiumActiveConcentration" 1e18 "GaN_con")

#- Contact Definitions
(sdegeo:define-contact-set "Top" "4" (color:rgb 1 0 0) "01")
(sdegeo:set-current-contact-set "Top")
(sdegeo:define-2d-contact (find-edge-id (position (/ LCon 2) 0 0))
(sdegeo:get-current-contact-set)); Top definition

(sdegeo:define-contact-set "Bottom" "4" (color:rgb 0 1 0) "10")
(sdegeo:set-current-contact-set "Bottom")
(sdegeo:define-2d-contact (find-edge-id (position (/ LD 2) (+ WB 7) 0))
(sdegeo:get-current-contact-set)); Bttom definition

#--- Meshing Strategy 
(sdedr:define-refinement-window "RefEvalWin.all" "Rectangle" (position 0 0 0) (position LD (+ WB 7) 0)) ; define refinement window(all)
(sdedr:define-refinement-size "Mesh_all" 10 10 10 10)
(sdedr:define-refinement-placement "Placement_all" "Mesh_all" "RefEvalWin.all")

(sdedr:define-refinement-window "RefEvalWin.Top" "Rectangle" (position 0 0 0) (position LD 0.01 0)) ; define refinement window(all)
(sdedr:define-refinement-size "Mesh_Top" 10 2.5e-3 10 2.5e-3)
(sdedr:define-refinement-placement "Placement_Top" "Mesh_Top" "RefEvalWin.Top")

(sdedr:define-refinement-window "RefEvalWin.Top2" "Rectangle" (position 0 0.01 0) (position LD 0.1 0)) ; define refinement window(all)
(sdedr:define-refinement-size "Mesh_Top2" 10 1e-2 10 1e-2)
(sdedr:define-refinement-placement "Placement_Top2" "Mesh_Top2" "RefEvalWin.Top2")

(sdedr:define-refinement-window "RefEvalWin.Top3" "Rectangle" (position 0 0.1 0) (position LD 0.5 0)) ; define refinement window(all)
(sdedr:define-refinement-size "Mesh_Top3" 10 0.1 10 0.1)
(sdedr:define-refinement-placement "Placement_Top3" "Mesh_Top3" "RefEvalWin.Top3")

(sdedr:define-refinement-window "RefEvalWin.Top4" "Rectangle" (position 0 0.5 0) (position LD 5 0)) ; define refinement window(all)
(sdedr:define-refinement-size "Mesh_Top4" 10 0.5 10 0.5)
(sdedr:define-refinement-placement "Placement_Top4" "Mesh_Top4" "RefEvalWin.Top4")

#--- Build Mesh
(sde:build-mesh "snmesh" " -a " "Ga2O3_GaN_SBD_Nd@Nd@_GW@GW@_msh.tdr")
